START
    FOR r = 0 TO 8 DO
        // 1. Gambar BOX (9x9)
        FOR c = 0 TO 8 DO
            IF r == 0 OR r == 8 OR c == 0 OR c == 8 THEN
                PRINT "*" (without newline)
            ELSE
                PRINT " " (without newline)
            ENDIF
        ENDFOR
        PRINT "   " (without newline)

        // 2. Gambar OVAL (9x9)
        FOR c = 0 TO 8 DO
            IF (r == 0 OR r == 8) AND (c >= 3 AND c <= 5) THEN
                PRINT "*" (without newline)
            ELSE IF (r == 1 OR r == 7) AND (c == 1 OR c == 7) THEN
                PRINT "*" (without newline)
            ELSE IF (r >= 2 AND r <= 6) AND (c == 0 OR c == 8) THEN
                PRINT "*" (without newline)
            ELSE
                PRINT " " (without newline)
            ENDIF
        ENDFOR
        PRINT "   " (without newline)

        // 3. Gambar ARROW / PANAH (9x9)
        FOR c = 0 TO 8 DO
            IF r == 0 AND c == 4 THEN
                PRINT "*" (without newline)
            ELSE IF r == 1 AND (c >= 3 AND c <= 5) THEN
                PRINT "*" (without newline)
            ELSE IF r == 2 AND (c >= 2 AND c <= 6) THEN
                PRINT "*" (without newline)
            ELSE IF r >= 3 AND c == 4 THEN
                PRINT "*" (without newline)
            ELSE
                PRINT " " (without newline)
            ENDIF
        ENDFOR
        PRINT "   " (without newline)

        // 4. Gambar DIAMOND (9x9)
        FOR c = 0 TO 8 DO
            IF ABS(r - 4) + ABS(c - 4) == 4 THEN
                PRINT "*" (without newline)
            ELSE
                PRINT " " (without newline)
            ENDIF
        ENDFOR

        PRINT NEWLINE
    ENDFOR
END
